# The whole game on a careless route: the cigarette left behind, too slow to
# leave the campsite, the noisy cabinet, seen in the factory, spotted three
# times, X not kept breathing (the Reaper takes him), every timed choice
# run out, no cigarette left for the summit. Losing a fight, or X's
# condition running out, is death: the level starts again ("died", "retry"),
# and the second try gets through.
#   "While It Lasts" --walkthrough tests/walkthrough/careless.txt
# The game must still reach its end, with the choices visible in it.

# --- Stage 00 ---------------------------------------------------------------
room campsite
use cigarette_pack
choose Leave
expect cig_count == 0
talk                          # no bandage yet
expect x_woken && !wound_dressed
use medical_kit
talk
choose Wait                   # not yet
talk
choose Apply
expect fire_out
wait 34                       # too slow: branches snap, then the silhouette
expect x_condition == 2
use forest_exit

# --- Stage 01 ---------------------------------------------------------------
room factory_entrance
use door
room factory_office
use first_aid_cabinet
choose Force it
expect noisy_entry && cabinet_open && cig_count == 1 && bandages == 2
use floor_door
room factory_floor
walk 110
choose Sneak                  # it heard the cabinet: seen, and it charges
fight lose                    # Y goes down, and with no health left he dies...
died
retry                         # ...and the floor starts again, as it was
room factory_floor
expect !factory_won && y_health == 5 && noisy_entry
walk 110
choose Sneak                  # seen again; this time it is fought
fight
expect factory_won
use exit_office
room factory_office
use exit_entrance
room factory_entrance
talk
choose Share
expect !has_antibiotics && x_condition == 2 && cig_count == 0 && smoked_with_x == 1 && bandages == 1
use road

# --- Stage 02 ---------------------------------------------------------------
room checkpoint
wait 30                       # standing in the open: spotted each time it watches,
                              # put back at the start; the second time costs
                              # X_CONDITION, the third is the chase...
choose timeout                # ...and too slow to run: X has nothing left
died
retry
room checkpoint
expect spotted_count == 0 && x_condition == 2
use bottles
item bottle                   # this time a bottle into the trees
choose Throw
use exit_road
room barricade
use rod
use slab
choose Lever
fight lose                    # the crawlers get Y: dead again
died
retry
room barricade
use rod
use slab
choose Lever
fight
revive                        # kept breathing this time
expect reaper_touched == 0 && x_condition == 1   # a crawler got to him in the fight
grab
use exit_road

# --- Stage 03 ---------------------------------------------------------------
room foot
use crushed_pack
choose Smoke
expect smoked_with_x == 2 && cig_count == 0
use lantern                   # X's line for a low condition
use steps
room collapse
revive slow                   # nobody keeps him breathing: the Reaper takes him
died
retry
room collapse
revive
expect reaper_touched == 0 && dragging && x_condition == 1
walk 200
use steps
room broken_stairs
walk 105
grab
walk 200                      # walking up to it starts the fight
fight lose
died
retry
room broken_stairs
walk 105
grab
walk 200
fight
expect creature_cleared
use fallen_beam
talk
use steps
room final_ascent
walk 160
use summit_path

# --- Stage 04 ---------------------------------------------------------------
room summit
talk                          # no cigarette left: X's own falls out of the blanket
expect cig_scene_done && cig_lit
wait 31                       # nothing touched: the Reaper comes anyway
ended
